cmake_minimum_required(VERSION 3.21)
project(AramfWorkerStation VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets)
add_executable(AramfWorkerStation WIN32 src/main.cpp)
target_link_libraries(AramfWorkerStation PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets)
if(MINGW)
  target_compile_options(AramfWorkerStation PRIVATE -Wall -Wextra -Wpedantic)
endif()
